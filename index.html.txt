      const iconEl = document.getElementById('toastIcon');

      titleEl.textContent = title;

    function selectSlot(slotId) {

      event.stopPropagation();
      delete roster[slotId];
      renderAll();

    function clearAllSlots() {
      roster = {};
      selectedReelCandidate = null;
      renderAll();
      showToast('Roster Cleared', 'All 8 administration slots are open and your $100M cap is restored.', 'emerald');
    }

    function draftCandidate(name) {
      const cand = CANDIDATES.find(c => c.name === name);
      if (!cand) return;

      // Prevent duplicate drafting of the same person into two slots
      const existingSlot = Object.entries(roster).find(([slotId, c]) => c.name === name && slotId !== selectedSlot);
      if (existingSlot) {
        const slotTitle = SLOTS.find(s => s.id === existingSlot[0]).title;
        showToast('Already on Roster', `${cand.name} is already serving as your ${slotTitle}. Remove them first to reassign.`);
        return;
      }

      const spentExcludingCurrent = Object.entries(roster).reduce(
        (sum, [slotId, c]) => (slotId === selectedSlot ? sum : sum + c.cost),
        0
      );

      if (spentExcludingCurrent + cand.cost > 100) {
        showToast('Over $100M Cap', `Drafting ${cand.name} ($${cand.cost}M) exceeds your remaining Political Capital cap.`);
        return;
      }

      roster[selectedSlot] = cand;

      // Advance selectedSlot automatically to the next open slot for smooth drafting
      const nextOpen = SLOTS.find(s => !roster[s.id]);
      if (nextOpen) selectedSlot = nextOpen.id;

      renderAll();
      loadReel(cand.name);
    }

    function autoFillConsensus() {
      roster = {};
      SLOTS.forEach(s => {
        const match = CANDIDATES.find(c => c.name === s.americasPick);
        if (match) roster[s.id] = match;
      });
      renderAll();
      loadReel(SLOTS[0].americasPick);
      showToast("The People's House Loaded", "Drafted all 8 of the nation's #1 most-selected officials ($99M total cap).", 'emerald');
    }

    function lock2028Ballot() {
      const count = Object.keys(roster).length;
      const el = document.getElementById('lockStatus');
      el.classList.remove('hidden');

      if (count < 8) {
        el.className = 'text-xs font-semibold text-amber-400';
        el.textContent = `Fill all 8 slots first (${count}/8 drafted) to timestamp your 2028 Dream House ticket.`;
        showToast('Incomplete Roster', `You have ${count}/8 slots filled. Complete your lineup to lock your 2028 capsule.`);

        const timestamp = new Date().toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
        el.className = 'text-xs font-semibold text-emerald-400';
        el.textContent = `✓ Locked on ${timestamp} for the Nov 2028 & Jan 2029 confirmation window!`;
        showToast('2028 Capsule Locked', 'Your Dream House roster is timestamped for the 2028 election cycle.', 'emerald');
      }
    }

      document.getElementById('broadcastQuote').textContent = perspectives[currentNetwork].quote;

      document.getElementById('civicLessonBox').classList.remove('hidden');
      document.getElementById('civicLessonText').textContent = cand.lesson;

      renderAll();

    function setNetwork(net) {
      currentNetwork = net;
      ['CSPAN', 'CNN', 'FOX'].forEach(n => {
        const btn = document.getElementById(`net-${n}`);
        btn.className = n === net
          ? 'px-3 py-1.5 text-xs font-bold rounded-lg bg-blue-600 text-white transition'
          : 'px-3 py-1.5 text-xs font-bold rounded-lg text-slate-400 hover:text-white transition';
      });
      if (selectedReelCandidate) loadReel(selectedReelCandidate);
    }

      ['draft', 'consensus', 'kitchen', 'reel'].forEach(t => {
        document.getElementById(`view-${t}`).classList.toggle('hidden', t !== tab);
        const btn = document.getElementById(`tab-${t}`);

      });
